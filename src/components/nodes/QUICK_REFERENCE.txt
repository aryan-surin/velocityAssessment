// ============================================================================
// MINIMAL NODE EXAMPLE
// ============================================================================

import { BaseNode } from './BaseNode';
import { createNodeConfig, createHandle, createField } from './nodeConfig';

const simpleNodeConfig = createNodeConfig({
  title: 'Simple Node',
  handles: [
    createHandle('target', 'input'),
    createHandle('source', 'output')
  ],
  fields: [
    createField('value', 'Value', 'text')
  ]
});

export const SimpleNode = ({ id, data }) => {
  return <BaseNode id={id} data={data} config={simpleNodeConfig} />;
};

// ============================================================================
// FIELD TYPE EXAMPLES
// ============================================================================

// Text Input
createField('name', 'Name', 'text', {
  defaultValue: 'Default',
  placeholder: 'Enter name'
})

// Textarea
createField('description', 'Description', 'textarea', {
  defaultValue: '',
  placeholder: 'Enter description',
  rows: 4
})

// Select Dropdown
createField('type', 'Type', 'select', {
  defaultValue: 'option1',
  options: [
    { value: 'option1', label: 'Option 1' },
    { value: 'option2', label: 'Option 2' }
  ]
})

// Number Input
createField('count', 'Count', 'number', {
  defaultValue: 0,
  min: 0,
  max: 100,
  step: 1
})

// Checkbox
createField('enabled', 'Enabled', 'checkbox', {
  defaultValue: false
})

// ============================================================================
// HANDLE CONFIGURATIONS
// ============================================================================

// Single input, single output (most common)
handles: [
  createHandle('target', 'input'),
  createHandle('source', 'output')
]

// Multiple inputs, single output
handles: [
  createHandle('target', 'input1', { style: { top: '30%' } }),
  createHandle('target', 'input2', { style: { top: '60%' } }),
  createHandle('source', 'output')
]

// Single input, multiple outputs
handles: [
  createHandle('target', 'input'),
  createHandle('source', 'output1', { style: { top: '40%' } }),
  createHandle('source', 'output2', { style: { top: '70%' } })
]

// No inputs (data source)
handles: [
  createHandle('source', 'output')
]

// No outputs (data sink)
handles: [
  createHandle('target', 'input')
]

// ============================================================================
// STYLING EXAMPLES
// ============================================================================

// Color coding by category
style: {
  backgroundColor: '#e3f2fd'  // Blue - Input nodes
}

style: {
  backgroundColor: '#fff3e0'  // Orange - Output nodes
}

style: {
  backgroundColor: '#f3e5f5'  // Purple - AI/LLM nodes
}

style: {
  backgroundColor: '#e8f5e9'  // Green - Text/Data nodes
}

style: {
  backgroundColor: '#fce4ec'  // Pink - Filter/Transform nodes
}

style: {
  backgroundColor: '#fff9c4'  // Yellow - Logic/Conditional nodes
}

// Custom dimensions
style: {
  width: 250,
  minHeight: 120
}

// Custom borders
style: {
  border: '2px solid #1976d2',
  borderRadius: '8px'
}
